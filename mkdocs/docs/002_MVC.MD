## Example

```java
@Slf4j
@RestController
@SuppressWarnings("ALL")
@RequestMapping(path = "/authorize")
public class AuthorizeController extends AppController {
    
    @Aautowire
    AuthorizeService authorizeService;

    // embed converter
    @PostMapping("/login")
    public ResponseEntity login(@ConverterResolver AuthorizeConverter converter, @Valid @RequestBody AuthorizeRequest request) {
        ServiceResolver resolver = authorizeService.login(manifest);
        return response(resolver, converter);
    }

    @Autowire
    AuthorizeConverter converterAtuowire;

    // declare autowired converter
    @PostMapping("/login")
    public ResponseEntity login(@Valid @RequestBody AuthorizeRequest request) {
        ServiceResolver resolver = authorizeService.login(manifest);
        return response(resolver, converterAtuowire);
    }
    
}
```
